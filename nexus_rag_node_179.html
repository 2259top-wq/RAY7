# ---------------------------------------------------------
# 【RAG 向量檢索架構實作】: BM25 - 實作細節 3 的核心流程
# 展示 Document -> Embedding -> Vector DB -> Generation 的完整管線

from langchain.embeddings import OpenAIEmbeddings
from langchain.vectorstores import Chroma
from langchain.prompts import ChatPromptTemplate
from langchain.chat_models import ChatOpenAI

class RAGPipeline:
    def __init__(self, documents: list[str]):
        # 步驟 1: 將文字轉為向量 (Embedding) 並存入向量資料庫
        self.embeddings = OpenAIEmbeddings()
        self.vector_store = Chroma.from_texts(
            texts=documents, 
            embedding=self.embeddings
        )
        self.retriever = self.vector_store.as_retriever(search_kwargs={"k": 3})
        self.llm = ChatOpenAI(model="gpt-3.5-turbo")
        
    def answer_question(self, query: str) -> str:
        # 步驟 2: 檢索 (Retrieval) - 找出最相似的 Top-K 文件
        docs = self.retriever.get_relevant_documents(query)
        context = "
".join([doc.page_content for doc in docs])
        
        # 步驟 3: 生成 (Generation) - 將 Context 注入 Prompt
        prompt_template = ChatPromptTemplate.from_template(
            "請根據以下參考資料回答問題：
{context}

問題：{query}"
        )
        messages = prompt_template.format_messages(context=context, query=query)
        
        print(f"[*] 已檢索到 {len(docs)} 篇相關文章，正在生成解答...")
        return self.llm(messages).content
