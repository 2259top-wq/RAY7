import torch
import torch.nn as nn

class MultimodalCrossAttention(nn.Module):
    def __init__(self, visual_dim, text_dim):
        super().__init__()
        # 將不同模態的特徵投影到相同的潛在空間 (Latent Space)
        self.vision_proj = nn.Linear(visual_dim, 512)
        self.text_proj = nn.Linear(text_dim, 512)
        
        self.attention = nn.MultiheadAttention(embed_dim=512, num_heads=8)
        
    def forward(self, image_features, text_embeddings):
        # 步驟 1: 空間投影對齊
        v_feat = self.vision_proj(image_features)
        t_feat = self.text_proj(text_embeddings)
        
        # 步驟 2: Cross-Attention
        # Query 來自影像，Key/Value 來自文字描述
        # 這使得模型能根據文字提示，聚焦於影像的特定區域
        attn_output, attn_weights = self.attention(
            query=v_feat,
            key=t_feat,
            value=t_feat
        )
        return attn_output
