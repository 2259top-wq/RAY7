import openai
from typing import List, Dict

class AgentExecutor:
    def __init__(self, tools: List[Dict]):
        self.tools = {t['name']: t['func'] for t in tools}
        self.memory = [] # 代理的長期/短期記憶庫
        
    def run(self, user_prompt: str) -> str:
        self.memory.append({"role": "user", "content": user_prompt})
        
        while True:
            # 步驟 1: Thought (思考) - 讓 LLM 決定下一步
            response = openai.ChatCompletion.create(
                model="gpt-4",
                messages=self.memory,
                functions=[t['schema'] for t in self.tools]
            )
            
            message = response.choices[0].message
            self.memory.append(message)
            
            # 步驟 2: Action (行動) - 如果 LLM 決定呼叫工具
            if message.get("function_call"):
                func_name = message.function_call.name
                func_args = eval(message.function_call.arguments)
                
                print(f"[*] Agent 決定執行工具: {func_name}")
                observation = self.tools[func_name](**func_args)
                
                # 步驟 3: Observation (觀察) - 將工具執行結果存入記憶
                self.memory.append({
                    "role": "function",
                    "name": func_name,
                    "content": str(observation)
                })
            else:
                # 如果沒有呼叫工具，代表 Agent 已經得出最終解答
                return message.content
